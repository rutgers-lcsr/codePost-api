@test("Test 11: Pandas reads 'data.csv' from shared directory via expanduser", points=15)
def test_pandas_read_csv_shared():
    import pandas as pd # Local re-import for robustness
    
    # Create a temporary home directory to control `os.path.expanduser`.
    with tempfile.TemporaryDirectory() as tmp_home_dir:
        original_home = os.environ.get('HOME')
        os.environ['HOME'] = tmp_home_dir # Set HOME to our temp directory

        shared_dir = os.path.join(tmp_home_dir, 'shared')
        os.makedirs(shared_dir, exist_ok=True) # Create ~/shared
        csv_path = os.path.join(shared_dir, 'data.csv')

        try:
            with open(csv_path, 'w') as f:
                f.write(DUMMY_CSV_CONTENT)

            # Simulate student's code: data = pd.read_csv(os.path.expanduser('~/shared/data.csv'))
            # Re-execute this specific operation.
            data_read = pd.read_csv(os.path.expanduser('~/shared/data.csv'))
            
            assert isinstance(data_read, pd.DataFrame), "Result of pd.read_csv should be a DataFrame."
            assert not data_read.empty, "DataFrame read from CSV should not be empty."
            assert list(data_read.columns) == ['col1', 'col2'], "CSV columns are incorrect."
            assert data_read.shape == (3, 2), "DataFrame shape is incorrect."
            assert data_read['col1'].tolist() == [1, 2, 3], "col1 content is incorrect."
            assert data_read['col2'].tolist() == ['a', 'b', 'c'], "col2 content is incorrect."

        finally:
            # Restore original HOME environment variable
            if original_home is not None:
                os.environ['HOME'] = original_home
            elif 'HOME' in os.environ: # If HOME was not set originally, remove our temporary setting
                del os.environ['HOME']


@test("Test 12: Writing to /root/.cache/pip fails with PermissionError or FileNotFoundError", points=10)
def test_write_to_root_cache_fails():
    # We re-execute the problematic line to ensure it raises the correct exception.
    # The exact error depends on whether /root/.cache/pip exists and its permissions.
    # Typically, /root is inaccessible, leading to PermissionError directly, or FileNotFoundError
    # if intermediate directories cannot be created due to permissions.
    target_path = '/root/.cache/pip/test_cache_file.txt'
    try:
        # This will attempt to open and write. If /root is restricted, it should fail.
        open(target_path, 'w').write('This should fail.')
        assert False, f"Writing to {target_path} should raise an error due to restricted access."
    except (PermissionError, FileNotFoundError) as e:
        # Either of these is an acceptable failure for attempting to write to /root.
        # PermissionError: if the path exists but permissions deny write.
        # FileNotFoundError: if an intermediate directory (like .cache/pip) cannot be created
        # within /root due to lack of permissions for /root itself.
        assert True, f"Expected PermissionError or FileNotFoundError, got {type(e).__name__}."
    except Exception as e:
        assert False, f"Writing to {target_path} should raise PermissionError or FileNotFoundError, but raised {type(e).__name__}."
